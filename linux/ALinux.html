Nftables

● 表名为Skill2025,链名统一为标准链名的小写

● 默认情况下,表中所有的链默认策略均为DROP

● 始终允许来自172.16.10.0/24、172. 31.10.0/24、10.0.0.0/30、192. 168. 2.0/24网络的 流量

● 来自内网主机的所有流量中,应该只允许服务所需的流量通行

● 仅允许VPN客户端访问LAN1 网络

● 从internet网络发送到内网主机的所有流量中,应该只允许服务所需的流量通行

● 可以ping其他任何主机,但不允许被ping

apt-get install -y nftables

cat > /etc/nftables.conf <<'EOF'

#!/usr/sbin/nft -f

flush ruleset

table inet Skill2025 {

  chain input {

    type filter hook input priority filter; policy drop;

    iifname "lo" accept

    ct state established,related accept

    tcp dport 22 accept

    udp dport { 67, 1194 } accept

  }

  chain forward {

    type filter hook forward priority filter; policy drop;

    ct state established,related accept

    ip saddr 192.168.2.0/24 ip daddr 172.16.10.0/24 accept

    ip saddr { 10.0.0.0/30, 172.16.10.0/24, 172.31.10.0/24 } accept

    ip daddr 172.16.10.101 tcp dport { 80, 443, 999, 49000-50000 } accept

    ip daddr 172.16.10.102 tcp dport { 20, 990, 49000-50000 } accept

    ip daddr 172.31.10.0/24 udp dport 53 accept

  }

  chain output {

    type filter hook output priority filter; policy drop;

    oifname "lo" accept

    ct state established,related accept

    icmp type echo-request accept

    udp dport { 53, 67, 123, 1812, 1813 } accept

    tcp dport { 22, 53, 80, 443 } accept

  }

}

EOF

nft -c -f /etc/nftables.conf && nft -f /etc/nftables.conf

systemctl enable --now nftables

nft list table inet Skill2025 | grep policy

nft list chain inet Skill2025 input

nft list chain inet Skill2025 forward

nft list chain inet Skill2025 output

设备Host

Certificate Authority

•        配置根证书颁发机构

•        使用者名称: C=CN, O=RCA, CN=SKILL2025 Root CA - G4

•        CA目录：/etc/ssl/rca

•        CA证书以服务器的FQDN与扩展名(.key, .crt)命名

•        能够通过SERVER吊销列表服务器检测签发证书的有效性

•        本测试项目中所有应用所需的证书均由此根证书签发

根 CA

install -d -m 0700 /etc/ssl/rca/private

openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 \

  -out /etc/ssl/rca/private/host.skill2025.cn.key

chmod 600 /etc/ssl/rca/private/host.skill2025.cn.key

openssl req -x509 -new -sha256 -days 3650 \

  -key /etc/ssl/rca/private/host.skill2025.cn.key \

  -out /etc/ssl/rca/host.skill2025.cn.crt \

  -subj '/C=CN/O=RCA/CN=SKILL2025 Root CA - G4'

签发 APP1 证书

install -d -m 0700 /etc/ssl/rca/work

openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:2048 \

  -out /etc/ssl/rca/work/app1.skill2025.cn.key

openssl req -new -sha256 \

  -key /etc/ssl/rca/work/app1.skill2025.cn.key \

  -out /etc/ssl/rca/work/app1.skill2025.cn.csr \

  -subj '/C=CN/O=SKILL2025/CN=app1.skill2025.cn'

cat > /etc/ssl/rca/work/app1.ext <<'EOF'

basicConstraints=critical,CA:FALSE

keyUsage=critical,digitalSignature,keyEncipherment

extendedKeyUsage=serverAuth

subjectAltName=DNS:app1.skill2025.cn,DNS:skill2025.cn,DNS:*.skill2025.cn

EOF

openssl x509 -req -sha256 -days 825 \

  -in /etc/ssl/rca/work/app1.skill2025.cn.csr \

  -CA /etc/ssl/rca/host.skill2025.cn.crt \

  -CAkey /etc/ssl/rca/private/host.skill2025.cn.key -CAcreateserial \

  -extfile /etc/ssl/rca/work/app1.ext \

  -out /etc/ssl/rca/work/app1.skill2025.cn.crt

ssh root@172.16.10.101 'install -d -m 0755 /etc/ssl/tls'

scp /etc/ssl/rca/work/app1.skill2025.cn.key \

    /etc/ssl/rca/work/app1.skill2025.cn.crt \

    root@172.16.10.101:/etc/ssl/tls/

scp /etc/ssl/rca/host.skill2025.cn.crt \

    root@172.16.10.101:/etc/ssl/tls/skill2025-root-ca.crt

APP1验收

chmod 600 /etc/ssl/tls/app1.skill2025.cn.key
